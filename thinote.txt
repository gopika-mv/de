package com.example.exam;

import android.content.Intent;
import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.RadioButton;
import android.widget.RadioGroup;
import android.widget.Spinner;
import android.widget.TextView;
import android.widget.Toast;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class MainActivity extends AppCompatActivity {

    EditText bname, mname;
    RadioGroup rg;
    Spinner sp;
    Button b1;
    TextView price;
    String[] carType = {"Sedan", "Minivan", "SUV", "Hatchback"};
    String[] carPrices = {"₹11,00,000", "₹18,00,000", "₹15,50,000", "₹6,50,000"};
    String selectedprice = carPrices[0];

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        bname = findViewById(R.id.bname);
        mname = findViewById(R.id.mname);
        rg = findViewById(R.id.rg);
        sp = findViewById(R.id.sp);
        b1 = findViewById(R.id.b1);
        price = findViewById(R.id.price);

        ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_spinner_dropdown_item, carType);
        sp.setAdapter(adapter);

        sp.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                selectedprice = carPrices[position];
                price.setText("Price : " + selectedprice);
            }

            @Override
            public void onNothingSelected(AdapterView<?> parent) {
            }
        });

        b1.setOnClickListener(v -> {
            String brand = bname.getText().toString().trim();
            String model = mname.getText().toString().trim();
            int rbid = rg.getCheckedRadioButtonId();

            if (brand.isEmpty()) {
                bname.setError("Enter Brand Name");
                return;
            }
            if (model.isEmpty()) {
                mname.setError("Enter Model Name");
                return;
            }
            if (rbid == -1) {
                Toast.makeText(this, "Please select Fuel Type", Toast.LENGTH_SHORT).show();
                return;
            }

            RadioButton selectedButton = findViewById(rbid);
            String fueltype = selectedButton.getText().toString();
            String cartype = sp.getSelectedItem().toString();

            Intent i = new Intent(MainActivity.this, Sec2.class);
            i.putExtra("bname", brand);
            i.putExtra("mname", model);
            i.putExtra("fueltype", fueltype);
            i.putExtra("cartype", cartype);
            i.putExtra("price", selectedprice);
            startActivity(i);
        });
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.main_menu, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        if (item.getItemId() == R.id.showroom) {
            Intent i = new Intent(MainActivity.this, Sec2.class);
            startActivity(i);
            return true;
        }
        return super.onOptionsItemSelected(item);
    }
}

package com.example.exam;

import android.content.Intent;
import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

public class Sec2 extends AppCompatActivity {

    TextView tvBrand, tvModel, tvFuel, tvCarType, tvPrice;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_sec2);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        tvBrand = findViewById(R.id.tvBrand);
        tvModel = findViewById(R.id.tvModel);
        tvFuel = findViewById(R.id.tvFuel);
        tvCarType = findViewById(R.id.tvCarType);
        tvPrice = findViewById(R.id.tvPrice);

        Intent intent = getIntent();
        tvBrand.setText("Brand Name: " + intent.getStringExtra("bname"));
        tvModel.setText("Model Name: " + intent.getStringExtra("mname"));
        tvFuel.setText("Fuel Type: " + intent.getStringExtra("fueltype"));
        tvCarType.setText("Car Type: " + intent.getStringExtra("cartype"));
        tvPrice.setText("Price: " + intent.getStringExtra("price"));
    }

    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.main_menu, menu);
        return true;
    }

    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        if (item.getItemId() == R.id.showroom) {
            Intent i = new Intent(Sec2.this, MainActivity.class);
            startActivity(i);
            return true;
        }
        return super.onOptionsItemSelected(item);
    }
}
